// Experimental whole-plane search v4. The frozen v3 remains an executable
// reference inside this translation unit; no v3 or shared-core file is edited.
#define RATIONAL_SQUARE_V3_NO_MAIN
#include "all_plane_gap_v3.cpp"
#undef RATIONAL_SQUARE_V3_NO_MAIN

enum class GammaStage { late, table, rules, generation, lift };

static const char *gamma_stage_name(GammaStage stage) {
    switch(stage) {
        case GammaStage::late:return "late";
        case GammaStage::table:return "table";
        case GammaStage::rules:return "rules";
        case GammaStage::generation:return "generation";
        case GammaStage::lift:return "lift";
    }
    throw std::logic_error("unknown gamma stage");
}

struct GammaPowerPlan {
    struct Entry {u32 prime,power;unsigned exponent;};
    std::array<Entry,7> entries{};
    unsigned count=0;
    u32 relevant=1;
    explicit GammaPowerPlan(const CRTPlan &plan) {
        for(unsigned i=0;i<plan.count;++i) {
            const u32 p=plan.prime[i],power=plan.power[i];
            if(p%4!=1 || power==p) continue;
            unsigned e=0;for(u32 value=power;value>1;value/=p) ++e;
            // With the v3 critical h/u/v restrictions, gamma at 5^2 and
            // 13^2 is already 1. Other powers, including 5^3, are retained.
            if((p==5 || p==13) && e==2) continue;
            entries[count++]={p,power,e};
            relevant*=power;
        }
    }
};

struct GammaLabels {
    std::vector<u32> value;
    GammaLabels(u32 n,const GammaPowerPlan &plan) {
        if(!plan.count) return;
        value.assign(n+1,1);
        for(unsigned i=0;i<plan.count;++i) {
            const auto &f=plan.entries[i];
            u32 pk=1;
            for(unsigned k=1;k<f.exponent;++k) {
                pk*=f.prime;
                const u32 next=pk*f.prime;
                u32 factor=1;
                for(unsigned j=0;j<std::min(k,f.exponent-k);++j) factor*=f.prime;
                for(u32 t=pk;t<=n;t+=pk)
                    if(t%next) value[t]*=factor;
            }
        }
    }
    bool accepts(u32 u,u32 v,u32 h) const {
        return value.empty() || (value[u]==value[h] && value[v]==value[h]);
    }
};

struct GammaPowerRow {
    // type0: exact higher valuation, forbid 0/h modulo next.
    // type1: unit or divisible by the entire prime power.
    // type2: BOTH high-k branches are retained: exact e-k OR both exact k.
    struct Rule {unsigned type;u32 p,modulus,lower_next,high;};
    std::array<Rule,7> rules{};
    unsigned count=0;
    u32 h=0;
    GammaPowerRow(u32 difference,const GammaPowerPlan &plan):h(difference) {
        for(unsigned i=0;i<plan.count;++i) {
            const auto &f=plan.entries[i];
            unsigned k=0;u32 hk=1;
            while(k<f.exponent && h%(hk*f.prime)==0) {hk*=f.prime;++k;}
            if(k==0) continue; // Existing CRT already implies gamma=1.
            if(k==f.exponent) {
                rules[count++]={1,f.prime,f.power,0,0};
            } else if(2*k<=f.exponent) {
                u32 power=1;
                for(unsigned j=0;j<f.exponent-k+1;++j) power*=f.prime;
                rules[count++]={0,f.prime,power,0,0};
            } else {
                u32 lower=1;
                for(unsigned j=0;j<f.exponent-k+1;++j) lower*=f.prime;
                rules[count++]={2,f.prime,hk*f.prime,lower,hk};
            }
        }
    }
    bool accepts_rule(unsigned i,u32 u) const {
        const auto &r=rules[i];
        if(r.type==0) return u%r.modulus && (u-h)%r.modulus;
        if(r.type==1) return u%r.p || u%r.modulus==0;
        return u%r.lower_next || (u%r.high==0 && u%r.modulus && (u-h)%r.modulus);
    }
    bool accepts(u32 u,unsigned omit_mask=0) const {
        for(unsigned i=0;i<count;++i)
            if(!(omit_mask&(1u<<i)) && !accepts_rule(i,u)) return false;
        return true;
    }
};

static bool local_without19(const LocalSieve &local,u32 n,u32 u,u32 v) {
    return local.p5.accepts(n,u,v) && local.p11.accepts(n,u,v)
        && local.p7.accepts(n,u,v) && local.p29.accepts(n,u,v) && local.p37.accepts(n,u,v)
        && local.p13.accepts(n,u,v) && local.p17.accepts(n,u,v) && local.p23.accepts(n,u,v)
        && local.p31.accepts(n,u,v) && local.p41.accepts(n,u,v) && local.p43.accepts(n,u,v);
}

struct V4Stats:GapStats {
    u64 gamma_early_rejected=0,gamma_early_passed=0,gamma_lifted_progressions=0;
    u64 gamma_lift_residues=0,examined_generation_rejections=0;
};

template<class Callback>
static u64 generate_gamma_field_crt(u32 n,u32 h,const ArithmeticTables &tables,
                                   const ProgressionWheel<19> &wheel,const GammaPowerRow &gamma,
                                   bool lift,Callback emit,V4Stats &stats,u32 known_gcd) {
    const u32 m=n/known_gcd;
    const auto &plan=tables.plans[m];
    std::array<u32,128> residues{};
    unsigned count=1;
    for(unsigned i=0;i<plan.count;++i) {
        if(h%plan.power[i]==0) continue;
        const u32 shift=u32(u64(h)*plan.idempotent[i]%m);
        const unsigned old=count;
        if(2*old>residues.size()) throw std::logic_error("CRT capacity");
        for(unsigned j=0;j<old;++j) {
            u32 r=residues[j]+shift;
            if(r>=m) r-=m;
            residues[count++]=r;
        }
    }
    auto finish=[&](u32 first,u32 step,unsigned mask) {
        const auto *row=wheel.row(n,h,step);
        const unsigned all=(1u<<gamma.count)-1;
        for(u64 u=first;u<=n;u+=step) {
            const unsigned skip=row[u%19];
            if(skip==255) break;
            u+=u64(skip)*step;
            if(u>n) break;
            if(mask!=all && !gamma.accepts(u32(u),mask)) {
                ++stats.examined_generation_rejections;continue;
            }
            emit(u32(u));
        }
    };
    // A constraint depends only on u modulo its rule.modulus. Along one
    // progression its exact period is modulus/gcd(modulus,step). Lifting every
    // allowed residue is therefore exact, including both high-k branches.
    // Large periods or short progressions use direct rules to avoid expansion.
    auto refine=[&](auto &&self,u32 first,u32 step,unsigned index,unsigned mask)->void {
        if(index==gamma.count) {finish(first,step,mask);return;}
        const auto &rule=gamma.rules[index];
        const u32 period=rule.modulus/std::gcd(rule.modulus,step);
        const u32 length=(n-first)/step+1;
        if(!lift || period>61 || length<2*period) {
            self(self,first,step,index+1,mask);return;
        }
        ++stats.gamma_lifted_progressions;
        const u32 new_step=step*period; // lcm(step,modulus), a divisor of n.
        for(u32 j=0;j<period && u64(first)+u64(j)*step<=n;++j) {
            const u32 start=first+j*step;
            if(!gamma.accepts_rule(index,start)) continue;
            ++stats.gamma_lift_residues;
            self(self,start,new_step,index+1,mask|(1u<<index));
        }
    };
    for(unsigned i=0;i<count;++i) {
        u32 first=residues[i];
        if(first<h) first+=((h-first+m-1)/m)*m;
        if(first<=n) refine(refine,first,m,0,0);
    }
    return count;
}

template<class ParameterCallback,class PointCallback>
static V4Stats run_v4(const SearchOptions &opt,const ArithmeticTables &arithmetic,
                     const LocalSieve &local,const GapWheel &gapwheel,
                     const ProgressionWheel<19> &wheel,GammaStage stage,bool skip19,
                     ParameterCallback parameter,PointCallback accept) {
    V4Stats out;
    const u32 first_n=12*((opt.min_denominator+23)/24);
    for(u32 n=first_n;n<=opt.bound/2;n+=12) {
        const u32 N=2*n;
        ++out.exact.sides;
        out.dense_even_pairs+=u64(n+1)*(n+2)/2;
        out.gap_rows+=n+1;
        auto permitted=permitted_valuations(n,arithmetic.plans[n]);
        const CriticalContentRows critical_rows(arithmetic.plans[n]);
        critical_rows.mark(permitted,n);
        const GammaPowerPlan gamma_plan(arithmetic.plans[n]);
        const GammaLabels labels(stage==GammaStage::table?n:0,gamma_plan);
        const CriticalProjectionPlan critical_projection(arithmetic.plans[n]);
        gapwheel.generate(n,[&](u32 h) {
            ++out.local_gap_rows;++out.valuation_gap_rows;
            if(!permitted_difference(n,h,0,arithmetic)) return;
            ++out.difference_gap_rows;
            if(!critical_rows.accepts(h)) return;
            ++out.critical_gap_rows;
            const u32 gcd_nh=std::gcd(n,h);
            const GammaPowerRow gamma_row(h,gamma_plan);
            auto visit_pair=[&](u32 u) {
                const u32 v=u-h;
                ++out.generated_pairs;
                if(!permitted[u] || !permitted[v]) return;
                ++out.valuation_pairs;
                if((stage==GammaStage::table && !labels.accepts(u,v,h)) ||
                   (stage==GammaStage::rules && !gamma_row.accepts(u))) {
                    ++out.gamma_early_rejected;return;
                }
                ++out.gamma_early_passed;
                if(skip19 ? !local_without19(local,n,u,v) : !local.accepts(n,u,v)) return;
                ++out.local_pairs;
                if(!permitted_scale(n,u,v,gcd_nh)) return;
                ++out.scale_pairs;
                if(stage==GammaStage::late && gamma_plan.relevant>1) {
                    const u32 gh=gamma_value(gamma_plan.relevant,h);
                    if(gh!=gamma_value(gamma_plan.relevant,u) || gh!=gamma_value(gamma_plan.relevant,v)) {
                        ++out.gamma_rejected;return;
                    }
                }
                ++out.gamma_pairs;
                if(!critical_projection.accepts(u,h)) {++out.critical_projection_rejected;return;}
                ++out.critical_projection_pairs;
                parameter(N,2*u,2*v);
                evaluate(N,2*u,2*v,true,out.exact,[&](i64 x,i64 y,i64 a) {
                    accept(N,2*u,2*v,x,y,a,out.exact);
                });
            };
            if((stage==GammaStage::generation || stage==GammaStage::lift) && gamma_row.count)
                out.crt_residues+=generate_gamma_field_crt(n,h,arithmetic,wheel,gamma_row,
                    stage==GammaStage::lift,visit_pair,out,gcd_nh);
            else out.crt_residues+=generate_field_gap_crt(n,h,arithmetic,wheel,visit_pair,gcd_nh);
        });
    }
    return out;
}

int main(int argc,char**argv) try {
    SearchOptions opt;
    GammaStage stage=GammaStage::late;
    bool reference=false,skip19=true;
    std::string audit_path,parameter_path;
    for(int i=1;i<argc;++i) {
        const std::string arg=argv[i];
        auto value=[&]()->std::string {if(++i>=argc) throw std::runtime_error("missing option value");return argv[i];};
        if(arg=="--reference-v3") reference=true;
        else if(arg=="--keep-redundant19") skip19=false;
        else if(arg=="--gamma-stage") {
            const auto name=value();
            if(name=="late") stage=GammaStage::late;
            else if(name=="table") stage=GammaStage::table;
            else if(name=="rules") stage=GammaStage::rules;
            else if(name=="generation") stage=GammaStage::generation;
            else if(name=="lift") stage=GammaStage::lift;
            else throw std::runtime_error("gamma stage: late|table|rules|generation|lift");
        } else if(arg=="--audit-three") audit_path=value();
        else if(arg=="--audit-parameters") parameter_path=value();
        else if(arg=="--min-denominator") {
            const u64 lower=std::stoull(value());
            if(lower<1 || lower>2000000) throw std::runtime_error("minimum1..2000000");
            opt.min_denominator=u32(lower);
        } else if(arg=="--help") {
            std::cout<<"all_plane_gap_v4 [B] [--min-denominator L] [--audit-three CSV] "
                       "[--audit-parameters CSV] [--reference-v3] [--keep-redundant19] "
                       "[--gamma-stage late|table|rules|generation|lift]\n";return 0;
        } else {
            const u64 bound=std::stoull(arg);
            if(bound<1 || bound>2000000) throw std::runtime_error("bound1..2000000");
            opt.bound=u32(bound);
        }
    }
    if(opt.min_denominator>opt.bound) throw std::runtime_error("minimum exceeds bound");
    const auto started=std::chrono::steady_clock::now();
    ArithmeticTables arithmetic(std::max<u32>(1000,opt.bound/2));LocalSieve local;
    GapWheel gapwheel(local);ProgressionWheel<19> wheel(local);
    std::ofstream audit,parameters;
    if(!audit_path.empty()) {audit.open(audit_path);if(!audit)throw std::runtime_error("audit open");audit<<"N,p,q,X,Y,A\n";}
    if(!parameter_path.empty()) {parameters.open(parameter_path);if(!parameters)throw std::runtime_error("parameter audit open");parameters<<"N,p,q\n";}
    auto accept=[&](i64 N,i64 p,i64 q,i64 x,i64 y,i64 a,Stats &stats) {
        if(audit.is_open())audit<<N<<','<<p<<','<<q<<','<<x<<','<<y<<','<<a<<'\n';
        u64 d;if(square(u128(i128(N-x)*(N-x)+i128(N-y)*(N-y)),d)) {
            ++stats.solutions;
            std::cout<<"{\"solution\":{\"N\":"<<N<<",\"X\":"<<x<<",\"Y\":"<<y<<",\"A\":"<<a<<",\"D\":"<<d<<"}}\n";
        }
    };
    auto parameter=[&](u32 N,u32 p,u32 q) {if(parameters.is_open())parameters<<N<<','<<p<<','<<q<<'\n';};
    const auto search_started=std::chrono::steady_clock::now();
    V4Stats stats;
    if(reference) static_cast<GapStats&>(stats)=run_gap_search(opt,arithmetic,local,gapwheel,
        parameter,accept,false,true,true,&wheel,true);
    else stats=run_v4(opt,arithmetic,local,gapwheel,wheel,stage,skip19,parameter,accept);
    const auto finished=std::chrono::steady_clock::now();
    auto seconds=[](auto a,auto b){return std::chrono::duration<double>(b-a).count();};
    std::cout<<"{\"scope\":\"whole_plane\",\"generator\":\"gap_CRT_v4\""
             <<",\"max_minimal_coordinate_denominator\":"<<opt.bound
             <<",\"min_denominator\":"<<opt.min_denominator
             <<",\"reference_v3\":"<<(reference?"true":"false")
             <<",\"gamma_stage\":\""<<gamma_stage_name(stage)<<"\""
             <<",\"skip_redundant19\":"<<(skip19 && !reference?"true":"false")
             <<",\"sides\":"<<stats.exact.sides
             <<",\"dense_even_pairs\":"<<stats.dense_even_pairs
             <<",\"gap_rows\":"<<stats.gap_rows
             <<",\"local_gap_rows\":"<<stats.local_gap_rows
             <<",\"difference_gap_rows\":"<<stats.difference_gap_rows
             <<",\"critical_gap_rows\":"<<stats.critical_gap_rows
             <<",\"crt_residue_choices\":"<<stats.crt_residues
             <<",\"generated_pairs\":"<<stats.generated_pairs
             <<",\"valuation_pairs\":"<<stats.valuation_pairs
             <<",\"gamma_early_rejected\":"<<stats.gamma_early_rejected

             <<",\"gamma_lift_residues\":"<<stats.gamma_lift_residues
